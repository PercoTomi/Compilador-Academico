PROGRAMA
{
    long I, J, SUMA;
    I := 1L;
    SUMA := 0L;
    
    do
    {
        J := 1L;
        do
        {
            SUMA := SUMA + I * J;
            J := J + 1L;
        }
        until (J > 3L);
        I := I + 1L;
    }
    until (I > 3L);
    
    print(SUMA);
}
